// OLHO DE A VERDADE — SAMUCA × VOID let eyeTime = 0; let eyeBlink = 0; function drawHorrorEye(ctx, time) { const w = ctx.canvas.width; const h = ctx.canvas.height; eyeTime = time; const cx = w / 2; const cy = h / 2; ctx.save(); // Fundo completamente preto ctx.fillStyle = "#000"; ctx.fillRect(0, 0, w, h); // Tremor da câmera const shake = time > 80 ? Math.sin(time * 1.7) * 3 : 0; ctx.translate(shake, shake); // Veias pulsantes ctx.strokeStyle = "#8c0015"; ctx.lineWidth = 3; ctx.shadowColor = "#ff001e"; ctx.shadowBlur = 12; for (let i = 0; i < 18; i++) { const angle = i * Math.PI / 9; const pulse = Math.sin(time * 0.07 + i) * 12; ctx.beginPath(); ctx.moveTo(cx + Math.cos(angle) * 95, cy + Math.sin(angle) * 65); ctx.quadraticCurveTo( cx + Math.cos(angle) * (180 + pulse), cy + Math.sin(angle) * (140 + pulse), cx + Math.cos(angle) * 290, cy + Math.sin(angle) * 220 ); ctx.stroke(); } // Formato do olho const opening = Math.min(1, time / 65); const blink = time % 210 < 9 ? 0.08 : 1; const eyeHeight = 125 * opening * blink; ctx.shadowBlur = 35; ctx.fillStyle = "#a00019"; ctx.beginPath(); ctx.ellipse(cx, cy, 320, eyeHeight, 0, 0, Math.PI * 2); ctx.fill(); // Íris vermelha ctx.save(); ctx.beginPath(); ctx.ellipse(cx, cy, 310, Math.max(1, eyeHeight), 0, 0, Math.PI * 2); ctx.clip(); const iris = 55 + Math.sin(time * 0.08) * 5; const gradient = ctx.createRadialGradient( cx, cy, 5, cx, cy, iris * 2 ); gradient.addColorStop(0, "#050000"); gradient.addColorStop(0.25, "#250000"); gradient.addColorStop(0.55, "#ff001e"); gradient.addColorStop(1, "#65000e"); ctx.fillStyle = gradient; ctx.beginPath(); ctx.arc(cx, cy, iris * 2, 0, Math.PI * 2); ctx.fill(); // Pupila vertical ctx.shadowColor = "#ff001e"; ctx.shadowBlur = 25; ctx.fillStyle = "#030000"; ctx.beginPath(); ctx.ellipse( cx + Math.sin(time * 0.025) * 35, cy, 17, 80, 0, 0, Math.PI * 2 ); ctx.fill(); // Reflexo sobrenatural ctx.fillStyle = "#ffb5bd"; ctx.beginPath(); ctx.ellipse(cx - 35, cy - 32, 13, 5, -.5, 0, Math.PI * 2); ctx.fill(); ctx.restore(); // Bordas do olho ctx.strokeStyle = "#ff2946"; ctx.lineWidth = 8; ctx.shadowBlur = 30; ctx.beginPath(); ctx.ellipse(cx, cy, 320, Math.max(1, eyeHeight), 0, 0, Math.PI * 2); ctx.stroke(); // Mensagem final if (time > 100) { ctx.shadowBlur = 18; ctx.fillStyle = "#ff263e"; ctx.font = "bold 30px Arial"; ctx.textAlign = "center"; ctx.fillText("ISSO SÓ COMEÇOU.", cx, h - 65); } ctx.restore(); }
